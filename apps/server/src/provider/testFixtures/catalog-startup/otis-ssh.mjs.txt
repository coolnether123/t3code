import { createInterface } from "node:readline";

let attempts = 0;
createInterface({ input: process.stdin }).on("line", (line) => {
  const message = JSON.parse(line);
  if (message.id === undefined) return;
  let response;
  if (message.method === "initialize") {
    response = {
      result: {
        protocolVersion: "2025-11-25",
        capabilities: { tools: {} },
        serverInfo: { name: "startup-fixture", version: "1" },
      },
    };
  } else if (message.method === "tools/list") {
    attempts++;
    response =
      attempts < 3
        ? {
            error: {
              code: -32000,
              message:
                "Otis tool catalog unavailable. Retry tools/list; discovery will notify when the gateway recovers.",
            },
          }
        : { result: { tools: [{ name: "fixture_read", inputSchema: { type: "object" } }] } };
  } else {
    response = { error: { code: -32601, message: "Only discovery is supported by this fixture." } };
  }
  process.stdout.write(JSON.stringify({ jsonrpc: "2.0", id: message.id, ...response }) + "\n");
});
